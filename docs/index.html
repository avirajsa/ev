02

QUICK START

Initialize backend using Astral uv or Docker Compose stack.

# Setup environment and install dependencies
cp .env.example .env
uv venv
uv pip install -e ".[dev]"

# Start server
uv run uvicorn app.main:app --reload
# Launch backend, Postgres (pgvector), and Redis
docker compose up -d --build

# Check logs
docker compose logs -f ev-backend
make install    # Initialize venv & dependencies
make dev        # Run dev server with auto-reload
make test       # Execute pytest suite
make docker-up  # Boot Docker stack

{
  "type": "rpc_request",

  "action": "open_application",
